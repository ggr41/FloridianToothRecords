import java.util.*;

public class FloridianToothRecords {

    // --- Constants ---
    public static final int MAX_PEOPLE = 6;
    public static final int MAX_TEETH = 8;
    public static final int LAYERS = 2; // 0 = upper, 1 = lower

    // --- Data Structures ---
    private static String[] familyNames;
    private static char[][][] familyTeeth; // [person][layer][tooth]
    private static int numPeople;

    private static Scanner input = new Scanner(System.in);

    // --- Main Method ---
    public static void main(String[] args) {
        System.out.println("Welcome to the Floridian Tooth Records");

        numPeople = getNumberOfFamilyMembers();
        familyNames = new String[numPeople];
        familyTeeth = new char[numPeople][LAYERS][MAX_TEETH];

        inputFamilyData();

        char choice;
        do {
            displayMenu();
            choice = getMenuChoice();
            processChoice(choice);
        } while (choice != 'X');

        System.out.println("Exiting the Floridian Tooth Records :-)");
    }

    // --- Input Phase ---

    /** Prompts user for valid number of family members */
    public static int getNumberOfFamilyMembers() {
        int num;
        do {
            System.out.print("Please enter number of people in the family (1–6): ");
            while (!input.hasNextInt()) {
                System.out.println("Invalid input, try again.");
                input.next(); // clear buffer
            }
            num = input.nextInt();
            input.nextLine(); // consume newline

            if (num < 1 || num > MAX_PEOPLE) {
                System.out.println("Invalid number of people, try again.");
            }
        } while (num < 1 || num > MAX_PEOPLE);
        return num;
    }

    /** Collects names and teeth data for each family member */
    public static void inputFamilyData() {
        for (int i = 0; i < numPeople; i++) {
            System.out.print("Please enter the name for family member " + (i + 1) + ": ");
            familyNames[i] = input.nextLine().trim();

            for (int layer = 0; layer < LAYERS; layer++) {
                String layerName = (layer == 0) ? "uppers" : "lowers";
                String teethString;
                do {
                    System.out.print("Please enter the " + layerName + " for " + familyNames[i] + ": ");
                    teethString = input.nextLine().trim().toUpperCase();

                    if (!validateToothString(teethString)) {
                        System.out.println("Invalid tooth types, try again.");
                    }
                } while (!validateToothString(teethString));

                for (int t = 0; t < teethString.length(); t++) {
                    familyTeeth[i][layer][t] = teethString.charAt(t);
                }
                // Fill unused spots with 'M' (missing)
                for (int t = teethString.length(); t < MAX_TEETH; t++) {
                    familyTeeth[i][layer][t] = 'M';
                }
            }
        }
    }

    // --- Validation ---

    /** Checks that a tooth string is valid */
    public static boolean validateToothString(String s) {
        if (s.length() < 1 || s.length() > MAX_TEETH) return false;
        for (char c : s.toCharArray()) {
            if (c != 'I' && c != 'B' && c != 'M') return false;
        }
        return true;
    }

    // --- Menu System ---

    public static void displayMenu() {
        System.out.println("\n(P)rint, (E)xtract, (R)oot, e(X)it");
    }

    public static char getMenuChoice() {
        System.out.print("Enter your choice: ");
        String choice = input.nextLine().trim().toUpperCase();
        while (choice.length() != 1 || "PERX".indexOf(choice.charAt(0)) == -1) {
            System.out.print("Invalid menu option, try again: ");
            choice = input.nextLine().trim().toUpperCase();
        }
        return choice.charAt(0);
    }

    public static void processChoice(char choice) {
        switch (choice) {
            case 'P':
                printTeethRecord();
                break;
            case 'E':
                extractTooth();
                break;
            case 'R':
                computeRootCanals();
                break;
            case 'X':
                // handled in main loop
                break;
        }
    }

    // --- Option 1: Print ---

    public static void printTeethRecord() {
        for (int i = 0; i < numPeople; i++) {
            System.out.println("\n" + familyNames[i]);
            System.out.print("Uppers: ");
            for (int j = 0; j < MAX_TEETH; j++) {
                System.out.print((j + 1) + ":" + familyTeeth[i][0][j] + "  ");
            }
            System.out.print("\nLowers: ");
            for (int j = 0; j < MAX_TEETH; j++) {
                System.out.print((j + 1) + ":" + familyTeeth[i][1][j] + "  ");
            }
            System.out.println();
        }
    }

    // --- Option 2: Extract Tooth ---

    public static void extractTooth() {
        System.out.print("Which family member? ");
        String name = input.nextLine().trim();
        int personIndex = findFamilyMember(name);

        if (personIndex == -1) {
            System.out.println("Invalid family member, try again.");
            return;
        }

        System.out.print("Which tooth layer (U)pper or (L)ower? ");
        String layerInput = input.nextLine().trim().toUpperCase();
        int layerIndex = (layerInput.equals("U")) ? 0 : (layerInput.equals("L") ? 1 : -1);

        if (layerIndex == -1) {
            System.out.println("Invalid layer, try again.");
            return;
        }

        System.out.print("Which tooth number (1–8)? ");
        int toothNum;
        try {
            toothNum = Integer.parseInt(input.nextLine().trim());
        } catch (NumberFormatException e) {
            System.out.println("Invalid number, try again.");
            return;
        }

        if (toothNum < 1 || toothNum > MAX_TEETH) {
            System.out.println("Invalid tooth number, try again.");
            return;
        }

        if (familyTeeth[personIndex][layerIndex][toothNum - 1] == 'M') {
            System.out.println("Missing tooth, try again.");
            return;
        }

        familyTeeth[personIndex][layerIndex][toothNum - 1] = 'M';
        System.out.println("Tooth extracted successfully!");
    }

    public static int findFamilyMember(String name) {
        for (int i = 0; i < numPeople; i++) {
            if (familyNames[i].equalsIgnoreCase(name)) {
                return i;
            }
        }
        return -1;
    }

    // --- Option 3: Compute Root Canals ---

    public static void computeRootCanals() {
        int I = 0, B = 0, M = 0;

        for (int i = 0; i < numPeople; i++) {
            for (int layer = 0; layer < LAYERS; layer++) {
                for (int t = 0; t < MAX_TEETH; t++) {
                    char c = familyTeeth[i][layer][t];
                    if (c == 'I') I++;
                    else if (c == 'B') B++;
                    else if (c == 'M') M++;
                }
            }
        }

        System.out.println("Equation: " + I + "x² + " + B + "x - " + M + " = 0");

        double discriminant = (B * B) + (4 * I * M);
        double root1 = (-B + Math.sqrt(discriminant)) / (2.0 * I);
        double root2 = (-B - Math.sqrt(discriminant)) / (2.0 * I);

        System.out.printf("One root canal at %.2f%n", root1);
        System.out.printf("Another root canal at %.2f%n", root2);
    }
}
